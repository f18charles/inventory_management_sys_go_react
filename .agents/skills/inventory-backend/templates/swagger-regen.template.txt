// Makefile (excerpt) — regeneration is a workflow step, not automatic.

// swag init walks //@ annotation comments across the module starting from
// cmd/server/main.go and regenerates docs/docs.go + docs/swagger.json +
// docs/swagger.yaml. Handler annotations live in the domain packages
// (internal/<domain>/handler.go); swag discovers them from the general API
// info block in cmd/server/main.go.
// //@ annotations are inert on their own — swag does NOT run when you save.

swagger:
	cd backend && swag init -g cmd/server/main.go -o docs
