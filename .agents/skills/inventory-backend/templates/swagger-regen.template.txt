// Makefile (excerpt) — regeneration is a workflow step, not automatic.

// swag init parses //@ annotation comments starting from cmd/server/main.go
// and regenerates docs/docs.go + docs/swagger.json + docs/swagger.yaml.
// It does NOT run itself when you save a file.

swagger:
	swag init -g cmd/server/main.go -o docs
